-- running code
local function run(src)
	if not src or src:match("^%s*$") then notify("Nothing to execute", false) return false end
	local url = src:match("^%s*(https?://%S+)%s*$")
	if url then
		local body = httpGet(url)
		if not body then notify("Couldn't download that link", false) return false end
		src = body
	end
	local fn, e = loadstring(src)
	if not fn then warn("[shadow] " .. tostring(e)) notify("Syntax error, check output", false) return false end
	local ok, e2 = pcall(fn)
	if not ok then warn("[shadow] " .. tostring(e2)) notify("Script error, check output", false) return false end
	notify("Executed", true)
	return true
end

local classWords = {}

local function esc(s) return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")) end

local function highlight(src)
	local out, i, n, last = {}, 1, #src, ""
	local function put(color, s)
		last = s:sub(-1)
		out[#out + 1] = '<font color="' .. color .. '">' .. esc(s) .. "</font>"
	end
	while i <= n do
		local c = src:sub(i, i)
		if c == "\n" or c == " " or c == "\r" or c == "\t" then
			out[#out + 1] = c
			i = i + 1
		elseif src:sub(i, i + 1) == "--" then
			local eq = src:match("^%-%-%[(=*)%[", i)
			local e
			if eq then
				local _, ee = src:find("]" .. eq .. "]", i, true)
				e = ee or n
			else
				e = (src:find("\n", i, true) or n + 1) - 1
			end
			put(tok.cmt, src:sub(i, e)) i = e + 1
		elseif c == '"' or c == "'" then
			local j = i + 1
			while j <= n do
				local d = src:sub(j, j)
				if d == "\\" then j = j + 2
				elseif d == c then break
				elseif d == "\n" then j = j - 1 break
				else j = j + 1 end
			end
			j = math.min(j, n)
			put(tok.str, src:sub(i, j)) i = j + 1
		elseif src:match("^%[=*%[", i) then
			local eq = src:match("^%[(=*)%[", i)
			local _, ee = src:find("]" .. eq .. "]", i, true)
			local e = ee or n
			put(tok.str, src:sub(i, e)) i = e + 1
		elseif c:match("%d") then
			local s, e = src:find("^0[xX]%x+", i)
			if not s then s, e = src:find("^%d*%.?%d+[eE][+-]?%d+", i) end
			if not s then s, e = src:find("^%d*%.?%d*", i) end
			put(tok.num, src:sub(s, e)) i = e + 1
		elseif c:match("[%a_]") then
			local s, e = src:find("^[%w_]+", i)
			local w = src:sub(s, e)
			local nxt = src:match("^%s*(.)", e + 1)
			local color = tok.word
			if (last == "." or last == ":") then
				color = (nxt == "(") and tok.call or tok.word
			elseif flowWords[w] then color = tok.flow
			elseif kwWords[w] then color = tok.kw
			elseif classWords[w] then color = tok.cls
			elseif nxt == "(" or nxt == '"' or nxt == "'" then color = tok.call end
			put(color, w) i = e + 1
		else
			put(tok.op, c) i = i + 1
		end
	end
	return table.concat(out)
end

	table.sort(out, function(a, b)
		if a.exact ~= b.exact then return a.exact end
		local ra, rb = kindRank[a.kind] or 4, kindRank[b.kind] or 4
		if ra ~= rb then return ra < rb end
		if #a.label ~= #b.label then return #a.label < #b.label end
		return a.label < b.label
	end)
	return out
end

local function memberItems(list)
	local out = {}
	for _, m in ipairs(list) do
		local isFn = m:sub(-2) == "()"
		local name = isFn and m:sub(1, -3) or m
		out[#out + 1] = {label = name, kind = isFn and "fn" or "prop", insert = methodInserts[name]}
	end
	return out
end

local function docWords(skip)
	local seen, out = {}, {}
	local src = box.Text:sub(1, 20000)
	for w in src:gmatch("[%a_][%w_]*") do
		if #w >= 3 and w ~= skip and not seen[w] and not kwWords[w] then
			seen[w] = true
			out[#out + 1] = {label = w, kind = "word"}
		end
	end
	return out
end

-- works out what to suggest for the text before the caret
local function buildItems(before, manual)
	if before:find("%-%-") or before:find("%.%.%s*[%w_]*$") then return {}, 0 end
	local svc = before:match('GetService%(%s*["\']([%w_]*)$')
	if svc then
		local l = {}
		for _, s in ipairs(services) do l[#l + 1] = {label = s, kind = "cls", detail = "service"} end
		return rank(l, svc), #svc
	end
	local quotes = select(2, before:gsub('"', "")) + select(2, before:gsub("'", ""))
	if quotes % 2 == 1 then return {}, 0 end
	local obj, mem = before:match("([%a_][%w_]*)%s*[%.:]%s*([%w_]*)$")
	if obj then
		local list = members[obj]
		local items = memberItems(list or inst)
		if list and (obj == "game" or obj == "workspace" or obj == "Players" or obj == "LocalPlayer" or obj == "Character" or obj == "Humanoid") then
			for _, v in ipairs(memberItems(inst)) do items[#items + 1] = v end
		end
		return rank(items, mem), #mem
	end
	local word = before:match("([%a_][%w_]*)$") or ""
	if word == "" and not manual then return {}, 0 end
	if word:match("^%d") then return {}, 0 end
	local items = {}
	for _, k in ipairs(keywordList) do items[#items + 1] = {label = k, kind = "kw"} end
	for _, g in ipairs(globalList) do items[#items + 1] = {label = g[1], kind = g[2], detail = g[3]} end
	for _, s in ipairs(snippets) do items[#items + 1] = {label = s[1], kind = "snip", detail = s[2], body = s[3]} end
	for _, w in ipairs(docWords(word)) do items[#items + 1] = w end
	return rank(items, word), #word
end

function accept(it)
	local text, c = box.Text, box.CursorPosition

	local lineStart = base:match(".*\n()") or 1
	local indent = base:sub(lineStart):match("^%s*") or ""

	if not body and it.kind == "fn" and text:sub(c, c) ~= "(" then body = it.label .. "($0)" end
	body = body or it.label
	body = body:gsub("\n", "\n" .. indent)
	local at = body:find("$0", 1, true)
	body = body:gsub("%$0", "", 1)
	local newText = base .. body .. text:sub(c)
	local newC = #base + (at and at - 1 or #body) + 1
	hideSuggest()
	setText(newText, newC)
end

local function blockDepth(src)
	src = src:gsub("%-%-[^\n]*", ""):gsub('"[^"\n]*"', ""):gsub("'[^'\n]*'", "")
	local o, e = 0, 0
	for w in src:gmatch("[%a_][%w_]*") do
		if w == "function" or w == "if" or w == "for" or w == "while" then o = o + 1
		elseif w == "end" then e = e + 1 end
	end
	return o - e
end

local closers = {["("] = ")", ["["] = "]", ["{"] = "}", ['"'] = '"', ["'"] = "'"}
local closing = {[")"] = true, ["]"] = true, ["}"] = true, ['"'] = true, ["'"] = true}

local function newline(text, c)
	local before = text:sub(1, c - 2)
	local ls = before:match(".*\n()") or 1
	local line = before:sub(ls)
	local indent = line:match("^%s*")
	local tr_ = line:gsub("%s+$", "")
	local more = tr_:match("%f[%w_]then$") or tr_:match("%f[%w_]do$") or tr_:match("%f[%w_]else$") or tr_:match("%f[%w_]repeat$")
		or tr_:match("function%s*[%w_%.:]*%b()$") or tr_:match("[{(]$")
	local extra = more and "    " or ""
	local nxt = text:sub(c, c)
	local atEnd = nxt == "" or nxt == "\n"
	if tr_:match("[{(]$") and (nxt == "}" or nxt == ")") then
		return text:sub(1, c - 1) .. indent .. extra .. "\n" .. indent .. text:sub(c), c + #indent + #extra
	end
	local needsEnd = more and not tr_:match("[{(]$") and not tr_:match("%f[%w_]else$") and not tr_:match("%f[%w_]repeat$") and atEnd and blockDepth(text) > 0
	if needsEnd then
		return text:sub(1, c - 1) .. indent .. extra .. "\n" .. indent .. "end" .. text:sub(c), c + #indent + #extra
	end
	return text:sub(1, c - 1) .. indent .. extra .. text:sub(c), c + #indent + #extra
end

local function handleEdit()
	local text, c = box.Text, box.CursorPosition
	local prev = prevText
	if c < 1 or not box:IsFocused() then
		prevText = text
		hideSuggest()
		return
	end
	local typed
	if #text == #prev + 1 and c >= 2 and text:sub(1, c - 2) .. text:sub(c) == prev then typed = text:sub(c - 1, c - 1) end

	if (typed == "\n" or typed == "\t") and shown and items[sel] then
		setText(prev, c - 1)
		accept(items[sel])
		return
	end
	if typed == "\t" then
		setText(text:sub(1, c - 2) .. "    " .. text:sub(c), c + 3)
		hideSuggest()
		return
	end
	if typed == "\n" then
		local t2, c2 = newline(text, c)
		setText(t2, c2)
		hideSuggest()
		return
	end
	if typed and closing[typed] and text:sub(c, c) == typed then
		setText(text:sub(1, c - 2) .. text:sub(c), c)
		hideSuggest()
		return
	end
	if typed and closers[typed] then
		local pc = text:sub(c - 2, c - 2)
		local nc = text:sub(c, c)
		local quote = typed == '"' or typed == "'"
		if not (quote and (pc:match("[%w_]") or nc:match("[%w_]"))) and not nc:match("[%w_]") then
			setText(text:sub(1, c - 1) .. closers[typed] .. text:sub(c), c)
		else
			prevText = text
		end
		updateSuggest()
		return
	end
	if text:find("\t", 1, true) then
		local n = select(2, text:sub(1, c - 1):gsub("\t", ""))
		setText(text:gsub("\t", "    "), c + n * 3)
		hideSuggest()
		return
	end
	prevText = text
	if typed and (typed:match("[%w_]") or typed == "." or typed == ":" or typed == '"' or typed == "'") then
		updateSuggest()
	elseif shown then
		updateSuggest()
	else
		hideSuggest()
	end
end

local function toggleComment()
	local text, c = box.Text, box.CursorPosition
	if c < 1 then return end
	local ls = (text:sub(1, c - 1):match(".*\n()") or 1)
	local le = (text:find("\n", c, true) or #text + 1) - 1
	local line = text:sub(ls, le)
	local indent = line:match("^%s*")
	local rest = line:sub(#indent + 1)
	local newLine, delta
	if rest:sub(1, 2) == "--" then
		local stripped = rest:gsub("^%-%-%s?", "")
		newLine = indent .. stripped
		delta = #newLine - #line
	else
		newLine = indent .. "-- " .. rest
		delta = 3
	end
	setText(text:sub(1, ls - 1) .. newLine .. text:sub(le + 1), math.max(ls, c + delta))
end

	end

			local go_ = make("TextButton", {Size = UDim2.fromOffset(30, 30), Position = UDim2.fromOffset(234, 11), BackgroundColor3 = col.teal, Text = "", AutoButtonColor = false, BorderSizePixel = 0}, r)
			round(go_, 15)

local function C_10a()
    local script = G2L["10a"];
	local RunService = game:GetService("RunService")
	local UIS = game:GetService("UserInputService")
	local TweenService = game:GetService("TweenService")
	local Gui = G2L["1"]
	local Scripts = script.Parent

	local Util = require(Scripts:WaitForChild("Util"))
	local Anim = require(Scripts:WaitForChild("Anim"))
	local Nav = require(Scripts:WaitForChild("Nav"))
	local Editor = require(Scripts:WaitForChild("Editor"))
	local Cloud = require(Scripts:WaitForChild("Cloud"))
	local Config = require(Scripts:WaitForChild("Config"))
	local col = Util.col

	local conns = {}
	local function track(c) conns[#conns + 1] = c return c end
	Gui.Destroying:Connect(function()
		for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
	end)

	-- grab every uniquely named instance once
	local R = {gui = Gui, track = track}
	for _, d in ipairs(Gui:GetDescendants()) do
		if not d:IsDescendantOf(Scripts) and not d.Name:match("^Icon_") and R[d.Name] == nil then R[d.Name] = d end
	end
	R.pages = {home = R.HomePg, editor = R.EditorPg, hub = R.HubPg, config = R.ConfigPg}
	Util.paintIcons(Gui)

	Anim.setup(R)
	R.Anim = Anim
	Editor.setup(R)
	R.Editor = Editor
	Nav.setup(R)
	R.go = Nav.go
	Cloud.setup(R)
	Nav.onShow.hub = Cloud.hubShow
	Config.setup(R)

	-- toast
	do
		local toast, la = R.Toast, R.ToastLa
		local ts = toast:FindFirstChildOfClass("UIStroke")
		local token = 0
		Util.notify = function(msg, ok)
			token = token + 1
			local mine = token
			local bad = ok == false
			la.Text = msg
			la.TextColor3 = bad and col.red or col.teal
			ts.Color = bad and col.red or col.slate
			toast.Position = UDim2.new(0.5, 0, 1.08, 0)
			Util.tw(toast, {GroupTransparency = 0, Position = UDim2.new(0.5, 0, 1, -14)}, 0.3, Enum.EasingStyle.Quint)
			task.delay(2.2, function()
				if mine == token then Util.tw(toast, {GroupTransparency = 1}, 0.35) end
			end)
		end
	end

	-- slow aurora bands and the light sweep
	local bands = {
		{R.Band1, 0.2, 0.35, 0, -10},
		{R.Band2, 0.55, 0.27, 2.1, 8},
		{R.Band3, 0.88, 0.31, 4.2, -6},
	}
	track(RunService.Heartbeat:Connect(function()
		if not R.ShadowUI.Visible then return end
		local t = os.clock()
		for _, b in ipairs(bands) do
			b[1].Position = UDim2.new(0.5 + 0.07 * math.sin(t * b[3] + b[4]), 0, b[2] + 0.04 * math.sin(t * b[3] * 0.7 + b[4]), 0)
			b[1].Rotation = b[5] + 4 * math.sin(t * b[3] * 0.5 + b[4])
		end
	end))
	task.spawn(function()
		task.wait(2)
		while Gui.Parent do
			if R.ShadowUI.Visible then
				R.Sweep.Position = UDim2.new(-0.3, 0, 0.5, 0)
				local t = TweenService:Create(R.Sweep, TweenInfo.new(1.3, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Position = UDim2.new(1.3, 0, 0.5, 0)})
				t:Play()
				t.Completed:Wait()
			end
			task.wait(7)
		end
	end)

	-- close button in the sidebar just hides the ui
	local min = R.MinBn
	local minIc = min:FindFirstChild("Icon_close")
	min.MouseEnter:Connect(function() Util.tw(min, {BackgroundTransparency = 0.8}) if minIc then Util.tw(minIc, {Rotation = 90}, 0.2) end end)
	min.MouseLeave:Connect(function() Util.tw(min, {BackgroundTransparency = 0.92}) if minIc then Util.tw(minIc, {Rotation = 0}, 0.2) end end)
	min.MouseButton1Click:Connect(function() Anim.setOpen(false) end)

	-- floating button
	local fab = R.UIBn
	local fabStroke = fab:FindFirstChildOfClass("UIStroke")
	fab.MouseEnter:Connect(function() Util.tw(fab, {Size = UDim2.fromOffset(60, 60)}, 0.18) end)
	fab.MouseLeave:Connect(function() Util.tw(fab, {Size = UDim2.fromOffset(54, 54)}, 0.18) end)
	local drag, dStart, dPos, moved
	fab.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			drag, dStart, dPos, moved = true, i.Position, fab.Position, false
			Util.tw(fab, {Size = UDim2.fromOffset(48, 48)}, 0.1)
		end
	end)
	track(UIS.InputChanged:Connect(function(i)
		if drag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local d = i.Position - dStart
			if d.Magnitude > 6 then moved = true end
			if moved then fab.Position = UDim2.new(dPos.X.Scale, dPos.X.Offset + d.X, dPos.Y.Scale, dPos.Y.Offset + d.Y) end
		end
	end))
	track(UIS.InputEnded:Connect(function(i)
		if drag and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
			drag = false
			Util.tw(fab, {Size = UDim2.fromOffset(54, 54)}, 0.12)
			if not moved then Anim.setOpen(not Anim.isOpen()) end
		end
	end))
	track(RunService.Heartbeat:Connect(function()
		fabStroke.Transparency = 0.5 + 0.2 * math.sin(os.clock() * 2)
	end))
	track(UIS.InputBegan:Connect(function(i, gp)
		if gp then return end
		if i.KeyCode == Enum.KeyCode.RightShift then Anim.setOpen(not Anim.isOpen()) end
	end))

	-- profile picture (roblox can't load discord urls directly)
	task.spawn(function()
		if not (writefile and getcustomasset) then return end
		local ok, data = pcall(function() return game:HttpGet(Util.cfg.pfp, true) end)
		if ok and type(data) == "string" and #data > 100 then
			pcall(function() if makefolder and not isfolder("Shadow") then makefolder("Shadow") end end)
			pcall(writefile, "Shadow/pfp.png", data)
			local ok2, a = pcall(getcustomasset, "Shadow/pfp.png")
			if ok2 then R.Pfp.Image = a end
		end
	end)

	Nav.go("home")
	Cloud.loadHot()
	Anim.pop()
	print("[shadow] loaded - right shift or the ghost button toggles the ui")

end;
task.spawn(C_10a);

return G2L["1"], require;
